# (docs/0055 D12), rewritten for the redesigned header (docs/0082 D3,

# "as of" line survives, and flipping back reads the same count.

# Prerequisites: the same seeded fixture session as usage-seeded.txt
# (docs/0071 §4), named `freeze`, at 1280x800:
#   export DASHBOARD_GITHUB_FIXTURE=/work/crates/debug/verify/fixtures/github
#   export DASHBOARD_FIXED_NOW=2026-10-23T12:00:00Z
#   export DASHBOARD_GITHUB_USAGE_SEED=90
#   cargo run -q -p dashboard-debug -- session start --name freeze \
#     --size 1280x800 --binary /tmp/target/debug/my-dashboard
# Run (in-container): dashboard-debug run crates/debug/verify/usage-freeze.txt
#
# Every hop is a click on a fresh-boot accesskit id (audit.txt's
# walk-order note), never a frozen pixel coordinate: the old script's
# Home-card click broke when the layout moved. Labels are matched as
# normalized ("as of MMM D, HH:MM").
